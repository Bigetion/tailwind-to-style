import { register, cx } from 'tailwind-to-style/register'

// Define your button once...
register('btn', {
  base: { tw: 'px-5 py-2.5 rounded-lg font-semibold transition-all', 'font-family': 'inherit' },
  modifiers: {
    primary: { tw: 'bg-indigo-600 text-white hover:bg-indigo-700' },
    outline: { tw: 'bg-transparent border-2 border-indigo-600 text-indigo-600' },
    sm:      { tw: 'px-3 py-1.5', 'font-size': '12px' },
    lg:      { tw: 'px-7 py-3.5', 'font-size': '16px' },
    pill:    { 'border-radius': '9999px' },
  }
})

// ...use anywhere, like Bootstrap ↓

register() · register.group() · cx()

01 — register()

Define classes with modifiers

Use modifiers to auto-generate suffixed variants like .btn-primary, .btn-sm, .btn-pill — all from one config object.

Each modifier can mix Tailwind utilities via tw and raw CSS properties like 'font-size', 'border-radius', 'box-shadow'.

register() + modifiers 
register('btn', {
  base: {
    tw: 'rounded-lg font-semibold transition-all',
    'font-family': 'inherit',
    'line-height': '1',
  },
  modifiers: {
    // colors
    primary: { tw: 'bg-indigo-600 text-white hover:bg-indigo-700' },
    danger:  { tw: 'bg-red-600 text-white hover:bg-red-700' },
    // sizes
    sm:      { tw: 'px-3 py-1.5', 'font-size': '12px' },
    lg:      { tw: 'px-7 py-3.5', 'font-size': '16px' },
    // shapes
    pill:    { 'border-radius': '9999px' },
  }
})

// → .btn .btn-primary .btn-danger
// → .btn-sm .btn-lg .btn-pill
register.group() 
register.group('card', {
  root:   { tw: 'rounded-xl bg-white border shadow-sm' },
  header: {
    tw: 'px-6 py-4 border-b font-semibold',
    'font-size': '16px',
  },
  body:   { tw: 'px-6 py-4', 'line-height': '1.6' },
  footer: { tw: 'px-6 py-4 border-t bg-gray-50 text-sm' },
})

// → .card .card-header .card-body .card-footer

// Clean HTML:
<div class="card">
  <div class="card-header">Title</div>
  <div class="card-body">Content</div>
</div>

02 — register.group()

Multi-part components

Register related elements in one call. root maps to the base class name, everything else becomes .name-part.

Perfect for cards, modals, navbars, and any compound component. HTML stays clean — just class names, no inline styles.
